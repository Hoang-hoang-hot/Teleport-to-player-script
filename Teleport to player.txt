Instance.new("UICorner",main).CornerRadius = UDim.new(0,12)

local stroke = Instance.new("UIStroke")

layout.Parent = scroll
layout.Padding = UDim.new(0,5)

-- REFRESH
local function refresh()

	for _,v in pairs(scroll:GetChildren()) do
		if v:IsA("TextButton") then
			v:Destroy()
		end
	end

	for _,plr in pairs(Players:GetPlayers()) do
		if plr ~= LP then

			local card = Instance.new("TextButton")
			card.Parent = scroll
			card.Size = UDim2.new(1,-2,0,52)
			card.BackgroundColor3 = Color3.fromRGB(170,170,170)
			card.BackgroundTransparency = 0.55
			card.Text = ""

			Instance.new("UICorner",card).CornerRadius = UDim.new(0,10)

			local cardStroke = Instance.new("UIStroke")
			cardStroke.Parent = card
			cardStroke.Color = Color3.fromRGB(140,140,140)

			-- Avatar
			local avatar = Instance.new("ImageLabel")
			avatar.Parent = card
			avatar.Size = UDim2.new(0,38,0,38)
			avatar.Position = UDim2.new(0,7,0.5,-19)
			avatar.BackgroundTransparency = 1

			Instance.new("UICorner",avatar).CornerRadius = UDim.new(1,0)

			local img = Players:GetUserThumbnailAsync(
				plr.UserId,
				Enum.ThumbnailType.HeadShot,
				Enum.ThumbnailSize.Size100x100
			)

			avatar.Image = img

			-- Display Name
			local dname = Instance.new("TextLabel")
			dname.Parent = card
			dname.Position = UDim2.new(0,52,0,5)
			dname.Size = UDim2.new(1,-58,0,20)
			dname.BackgroundTransparency = 1
			dname.Text = plr.DisplayName
			dname.TextColor3 = Color3.new(0,0,0)
			dname.Font = Enum.Font.GothamBold
			dname.TextSize = 14
			dname.TextXAlignment = Enum.TextXAlignment.Left

			-- Username
			local uname = Instance.new("TextLabel")
			uname.Parent = card
			uname.Position = UDim2.new(0,52,0,24)
			uname.Size = UDim2.new(1,-58,0,16)
			uname.BackgroundTransparency = 1
			uname.Text = "@"..plr.Name
			uname.TextColor3 = Color3.fromRGB(50,50,50)
			uname.Font = Enum.Font.Gotham
			uname.TextSize = 11
			uname.TextXAlignment = Enum.TextXAlignment.Left

			-- TELEPORT
			card.MouseButton1Click:Connect(function()

				if plr.Character
				and plr.Character:FindFirstChild("HumanoidRootPart")
				and LP.Character
				and LP.Character:FindFirstChild("HumanoidRootPart") then

					LP.Character.HumanoidRootPart.CFrame =
						plr.Character.HumanoidRootPart.CFrame + Vector3.new(3,0,0)

				end
			end)
		end
	end
end

refresh()

Players.PlayerAdded:Connect(refresh)
Players.PlayerRemoving:Connect(refresh)

-- TOGGLE
local opened = false

toggle.MouseButton1Click:Connect(function()

	opened = not opened

	if opened then
		toggle.Text = "OFF"
		scroll.Visible = true
		main.Size = UDim2.new(0,260,0,320)
	else
		toggle.Text = "ON"
		scroll.Visible = false
		main.Size = UDim2.new(0,260,0,38)
	end
end)
